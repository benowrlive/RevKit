   155→
   156→export function ComparisonsPage() {
   157→  const review = useReviewStore((s) => s.review);
   158→  const addComparison = useReviewStore((s) => s.addComparison);
   159→  const deleteComparison = useReviewStore((s) => s.deleteComparison);
   160→  const renameComparison = useReviewStore((s) => s.renameComparison);
   161→  const deleteOutcome = useReviewStore((s) => s.deleteOutcome);
   162→  const addSubgroup = useReviewStore((s) => s.addSubgroup);
   163→  const renameSubgroup = useReviewStore((s) => s.renameSubgroup);
   164→  const deleteSubgroup = useReviewStore((s) => s.deleteSubgroup);
   165→
   166→  const [selectedOutcomeId, setSelectedOutcomeId] = useState<string | null>(null);
   167→  const [expandedComparisons, setExpandedComparisons] = useState<Set<string>>(
   168→    new Set(),
   169→  );
   170→
   171→  // Inline add-comparison form
   172→  const [newComparisonName, setNewComparisonName] = useState("");
   173→
   174→  // Add/Edit outcome dialog state
   175→  const [outcomeDialog, setOutcomeDialog] = useState<{
   176→    mode: "add" | "edit";
   177→    comparisonId?: string;
   178→    outcome?: Outcome;
   179→  } | null>(null);
   180→
   181→  // Pending delete (single AlertDialog handles all three kinds)
   182→  const [pendingDelete, setPendingDelete] = useState<
   183→    | { kind: "comparison" | "outcome" | "subgroup"; id: string; label: string }
   184→    | null
   185→  >(null);
   186→
   187→  // Inline rename state (comparison)
   188→  const [renamingComparison, setRenamingComparison] = useState<string | null>(
   189→    null,
   190→  );
   191→  const [renameComparisonValue, setRenameComparisonValue] = useState("");
   192→
   193→  // Inline add-subgroup form state (per outcome id)
   194→  const [addSubgroupFor, setAddSubgroupFor] = useState<string | null>(null);
   195→  const [newSubgroupName, setNewSubgroupName] = useState("");
   196→
   197→  // Inline rename subgroup
   198→  const [renamingSubgroup, setRenamingSubgroup] = useState<string | null>(null);
   199→  const [renameSubgroupValue, setRenameSubgroupValue] = useState("");
   200→
   201→  const comparisons = review?.comparisons ?? [];
   202→
   203→  // Find the selected outcome object (live from store so edits flow back).
   204→  const selectedOutcome = useMemo<Outcome | null>(() => {
   205→    if (!selectedOutcomeId) return null;
   206→    for (const c of comparisons) {
   207→      const o = c.outcomes.find((o2) => o2.id === selectedOutcomeId);
   208→      if (o) return o;
   209→    }
   210→    return null;
   211→  }, [comparisons, selectedOutcomeId]);
   212→
   213→  // Auto-expand the comparison that owns the selected outcome (handled in
   214→  // `handleSelectOutcome` below — no effect needed).
   215→  // Clear-selection-on-delete is also handled in the delete handler.
   216→
   217→  if (!review) return null;
   218→
   219→  function handleAddComparison() {
   220→    const name = newComparisonName.trim();
   221→    if (!name) {
   222→      toast.error("Name required");
   223→      return;
   224→    }
   225→    const id = addComparison(name);
   226→    setNewComparisonName("");
   227→    setExpandedComparisons((prev) => new Set(prev).add(id));
   228→  }
   229→
   230→  function confirmDelete() {
   231→    if (!pendingDelete) return;
   232→    if (pendingDelete.kind === "comparison") {
   233→      deleteComparison(pendingDelete.id);
   234→    } else if (pendingDelete.kind === "outcome") {
   235→      deleteOutcome(pendingDelete.id);
   236→      if (selectedOutcomeId === pendingDelete.id) setSelectedOutcomeId(null);
   237→    } else {
   238→      deleteSubgroup(pendingDelete.id);
   239→    }
   240→    toast.success("Deleted");
   241→    setPendingDelete(null);
   242→  }
   243→
   244→  function commitComparisonRename(id: string) {
   245→    const name = renameComparisonValue.trim();
   246→    if (!name) {
   247→      setRenamingComparison(null);
   248→      return;
   249→    }
   250→    renameComparison(id, name);
   251→    setRenamingComparison(null);
   252→  }
   253→
   254→  function commitSubgroupAdd(outcomeId: string) {
   255→    const name = newSubgroupName.trim();
   256→    if (!name) {
   257→      setAddSubgroupFor(null);
   258→      return;
   259→    }
   260→    addSubgroup(outcomeId, name);
   261→    setNewSubgroupName("");
   262→    setAddSubgroupFor(null);
   263→  }
   264→
   265→  function commitSubgroupRename(id: string) {
   266→    const name = renameSubgroupValue.trim();
   267→    if (!name) {
   268→      setRenamingSubgroup(null);
   269→      return;
   270→    }
   271→    renameSubgroup(id, name);
   272→    setRenamingSubgroup(null);
   273→  }
   274→
   275→  /** Select an outcome and ensure its parent comparison is expanded. */
   276→  function handleSelectOutcome(id: string, comparisonId: string) {
   277→    setSelectedOutcomeId(id);
   278→    setExpandedComparisons((prev) => {
   279→      if (prev.has(comparisonId)) return prev;
   280→      const next = new Set(prev);
   281→      next.add(comparisonId);
   282→      return next;
   283→    });
   284→  }
   285→
   286→  return (
   287→    <div className="space-y-4">
   288→      <div>
   289→        <h2 className="text-2xl font-bold tracking-tight">Comparisons & Outcomes</h2>
   290→        <p className="text-sm text-muted-foreground mt-1">
   291→          Build the analysis tree: comparisons → outcomes → subgroups. Enter data
   292→          per study on the right.
   293→        </p>
   294→      </div>
   295→
   296→      <div className="grid grid-cols-1 lg:grid-cols-[280px_1fr] gap-4 items-start">
   297→        {/* LEFT — Tree */}
   298→        <Card className="p-3 lg:sticky lg:top-4">
   299→          <div className="flex items-center justify-between mb-2">
   300→            <h3 className="text-sm font-semibold flex items-center gap-1.5">
   301→              <FolderTree className="size-4 text-emerald-600" />
   302→              Comparisons
   303→            </h3>
   304→          </div>
   305→
   306→          {/* Add comparison inline form */}
   307→          <div className="flex gap-1 mb-2">
   308→            <Input
   309→              value={newComparisonName}
   310→              onChange={(e) => setNewComparisonName(e.target.value)}
   311→              placeholder="New comparison name…"
   312→              className="h-8 text-sm"
   313→              onKeyDown={(e) => {
   314→                if (e.key === "Enter") handleAddComparison();
   315→                if (e.key === "Escape") setNewComparisonName("");
   316→              }}
   317→            />
   318→            <Button
   319→              size="icon"
   320→              variant="default"
   321→              className="h-8 w-8 bg-emerald-600 hover:bg-emerald-700 text-white shrink-0"
   322→              onClick={handleAddComparison}
   323→              title="Add comparison"
   324→            >
   325→              <Plus className="size-4" />
   326→            </Button>
   327→          </div>
   328→
   329→          <Separator className="mb-2" />
   330→
   331→          {comparisons.length === 0 ? (
   332→            <div className="text-center py-6 text-xs text-muted-foreground">
   333→              No comparisons yet.
   334→              <br />
   335→              Add one above to get started.
   336→            </div>
   337→          ) : (
   338→            <div className="space-y-0.5 max-h-[60vh] overflow-y-auto -mx-1 px-1">
   339→              {comparisons.map((c) => (
   340→                <ComparisonNode
   341→                  key={c.id}
   342→                  comparison={c}
   343→                  expanded={expandedComparisons.has(c.id)}
   344→                  onToggleExpand={() =>
   345→                    setExpandedComparisons((prev) => {
   346→                      const next = new Set(prev);
   347→                      if (next.has(c.id)) next.delete(c.id);
   348→                      else next.add(c.id);
   349→                      return next;
   350→                    })
   351→                  }
   352→                  selectedOutcomeId={selectedOutcomeId}
   353→                  onSelectOutcome={handleSelectOutcome}
   354→                  onAddOutcome={() =>
   355→                    setOutcomeDialog({
   356→                      mode: "add",
   357→                      comparisonId: c.id,
   358→                    })
   359→                  }
   360→                  onEditOutcome={(o) =>
   361→                    setOutcomeDialog({ mode: "edit", outcome: o })
   362→                  }
   363→                  onDeleteComparison={() =>
   364→                    setPendingDelete({
   365→                      kind: "comparison",
   366→                      id: c.id,
   367→                      label: c.name,
   368→                    })
   369→                  }
   370→                  onDeleteOutcome={(o) =>
   371→                    setPendingDelete({
   372→                      kind: "outcome",
   373→                      id: o.id,
   374→                      label: o.name,
   375→                    })
   376→                  }
   377→                  onRenameComparison={() => {
   378→                    setRenameComparisonValue(c.name);
   379→                    setRenamingComparison(c.id);
   380→                  }}
   381→                  renamingComparisonId={renamingComparison}
   382→                  renameComparisonValue={renameComparisonValue}
   383→                  onRenameComparisonValueChange={setRenameComparisonValue}
   384→                  onCommitComparisonRename={() => commitComparisonRename(c.id)}
   385→                  onCancelComparisonRename={() => setRenamingComparison(null)}
   386→                  addSubgroupFor={addSubgroupFor}
   387→                  newSubgroupName={newSubgroupName}
   388→                  onNewSubgroupNameChange={setNewSubgroupName}
   389→                  onStartAddSubgroup={(outcomeId) => {
   390→                    setAddSubgroupFor(outcomeId);
   391→                    setNewSubgroupName("");
   392→                  }}
   393→                  onCommitAddSubgroup={() => commitSubgroupAdd(addSubgroupFor ?? "")}
   394→                  onCancelAddSubgroup={() => setAddSubgroupFor(null)}
   395→                  onDeleteSubgroup={(sg) =>
   396→                    setPendingDelete({
   397→                      kind: "subgroup",
   398→                      id: sg.id,
   399→                      label: sg.name,
   400→                    })
   401→                  }
   402→                  renamingSubgroupId={renamingSubgroup}
   403→                  renameSubgroupValue={renameSubgroupValue}
   404→                  onRenameSubgroupValueChange={setRenameSubgroupValue}
   405→                  onStartRenameSubgroup={(sg) => {
   406→                    setRenameSubgroupValue(sg.name);
   407→                    setRenamingSubgroup(sg.id);
   408→                  }}
   409→                  onCommitRenameSubgroup={() => commitSubgroupRename(renamingSubgroup ?? "")}
   410→                  onCancelRenameSubgroup={() => setRenamingSubgroup(null)}
   411→                />
   412→              ))}
   413→            </div>
   414→          )}
   415→        </Card>
   416→
   417→        {/* RIGHT — Selected outcome detail */}
   418→        {!selectedOutcome ? (
   419→          <EmptyDetail comparisonsCount={comparisons.length} />
   420→        ) : (
   421→          <OutcomeDetail
   422→            outcome={selectedOutcome}
   423→            onEdit={() =>
   424→              setOutcomeDialog({ mode: "edit", outcome: selectedOutcome })
   425→            }
   426→            onAddSubgroup={() => {
   427→              setAddSubgroupFor(selectedOutcome.id);
   428→              setNewSubgroupName("");
   429→            }}
   430→            addSubgroupOpen={addSubgroupFor === selectedOutcome.id}
   431→            newSubgroupName={newSubgroupName}
   432→            onNewSubgroupNameChange={setNewSubgroupName}
   433→            onCommitAddSubgroup={() => commitSubgroupAdd(selectedOutcome.id)}
   434→            onCancelAddSubgroup={() => setAddSubgroupFor(null)}
   435→            renamingSubgroupId={renamingSubgroup}
   436→            renameSubgroupValue={renameSubgroupValue}
   437→            onRenameSubgroupValueChange={setRenameSubgroupValue}
   438→            onStartRenameSubgroup={(sg) => {
   439→              setRenameSubgroupValue(sg.name);
   440→              setRenamingSubgroup(sg.id);
   441→            }}
   442→            onCommitRenameSubgroup={() => commitSubgroupRename(renamingSubgroup ?? "")}
   443→            onCancelRenameSubgroup={() => setRenamingSubgroup(null)}
   444→            onDeleteSubgroup={(sg) =>
   445→              setPendingDelete({
   446→                kind: "subgroup",
   447→                id: sg.id,
   448→                label: sg.name,
   449→              })
   450→            }
   451→          />
   452→        )}
   453→      </div>
   454→
   455→      {/* Add / Edit outcome dialog */}
   456→      {outcomeDialog && (
   457→        <OutcomeFormDialog
   458→          mode={outcomeDialog.mode}
   459→          outcome={outcomeDialog.outcome}
   460→          comparisonId={outcomeDialog.comparisonId}
   461→          onClose={() => setOutcomeDialog(null)}
   462→        />
   463→      )}
   464→
   465→      {/* Delete confirmation */}
   466→      <AlertDialog
   467→        open={!!pendingDelete}
   468→        onOpenChange={(o) => {
   469→          if (!o) setPendingDelete(null);
   470→        }}
   471→      >
   472→        <AlertDialogContent>
   473→          <AlertDialogHeader>
   474→            <AlertDialogTitle>Delete {pendingDelete?.kind}?</AlertDialogTitle>
   475→            <AlertDialogDescription>
   476→              This will permanently delete{" "}
   477→              <span className="font-medium text-foreground">
   478→                {pendingDelete?.label}
   479→              </span>
   480→              {pendingDelete?.kind === "comparison" &&
   481→                " and all of its outcomes and data points."}
   482→              {pendingDelete?.kind === "outcome" &&
   483→                " and all of its data points and subgroups."}
   484→              {pendingDelete?.kind === "subgroup" &&
   485→                " and its data points."}
   486→              {" "}This cannot be undone.
   487→            </AlertDialogDescription>
   488→          </AlertDialogHeader>
   489→          <AlertDialogFooter>
   490→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   491→            <AlertDialogAction
   492→              onClick={confirmDelete}
   493→              className="bg-rose-600 hover:bg-rose-700 text-white"
   494→            >
   495→              Delete
   496→            </AlertDialogAction>
   497→          </AlertDialogFooter>
   498→        </AlertDialogContent>
   499→      </AlertDialog>
   500→    </div>
   501→  );
   502→}

   504→// --- Comparison tree node ----------------------------------------------------
   505→
   506→interface ComparisonNodeProps {
   507→  comparison: Comparison;
   508→  expanded: boolean;
   509→  onToggleExpand: () => void;
   510→  selectedOutcomeId: string | null;
   511→  onSelectOutcome: (id: string, comparisonId: string) => void;
   512→  onAddOutcome: () => void;
   513→  onEditOutcome: (o: Outcome) => void;
   514→  onDeleteComparison: () => void;
   515→  onDeleteOutcome: (o: Outcome) => void;
   516→  onRenameComparison: () => void;
   517→  renamingComparisonId: string | null;
   518→  renameComparisonValue: string;
   519→  onRenameComparisonValueChange: (v: string) => void;
   520→  onCommitComparisonRename: () => void;
   521→  onCancelComparisonRename: () => void;
   522→  addSubgroupFor: string | null;
   523→  newSubgroupName: string;
   524→  onNewSubgroupNameChange: (v: string) => void;
   525→  onStartAddSubgroup: (outcomeId: string) => void;
   526→  onCommitAddSubgroup: () => void;
   527→  onCancelAddSubgroup: () => void;
   528→  onDeleteSubgroup: (sg: { id: string; name: string }) => void;
   529→  renamingSubgroupId: string | null;
   530→  renameSubgroupValue: string;
   531→  onRenameSubgroupValueChange: (v: string) => void;
   532→  onStartRenameSubgroup: (sg: { id: string; name: string }) => void;
   533→  onCommitRenameSubgroup: () => void;
   534→  onCancelRenameSubgroup: () => void;
   535→}
   536→
   537→function ComparisonNode(props: ComparisonNodeProps) {
   538→  const {
   539→    comparison: c,
   540→    expanded,
   541→    onToggleExpand,
   542→    selectedOutcomeId,
   543→    onSelectOutcome,
   544→    onAddOutcome,
   545→    onEditOutcome,
   546→    onDeleteComparison,
   547→    onDeleteOutcome,
   548→    onRenameComparison,
   549→    renamingComparisonId,
   550→    renameComparisonValue,
   551→    onRenameComparisonValueChange,
   552→    onCommitComparisonRename,
   553→    onCancelComparisonRename,
   554→    addSubgroupFor,
   555→    newSubgroupName,
   556→    onNewSubgroupNameChange,
   557→    onStartAddSubgroup,
   558→    onCommitAddSubgroup,
   559→    onCancelAddSubgroup,
   560→    onDeleteSubgroup,
   561→    renamingSubgroupId,
   562→    renameSubgroupValue,
   563→    onRenameSubgroupValueChange,
   564→    onStartRenameSubgroup,
   565→    onCommitRenameSubgroup,
   566→    onCancelRenameSubgroup,
   567→  } = props;
   568→
   569→  const isRenaming = renamingComparisonId === c.id;
   570→
   571→  return (
   572→    <div className="rounded-md">
   573→      {/* Comparison header */}
   574→      <div className="flex items-center gap-1 group">
   575→        <button
   576→          onClick={onToggleExpand}
   577→          className="p-1 rounded hover:bg-muted text-muted-foreground shrink-0"
   578→          aria-label={expanded ? "Collapse" : "Expand"}
   579→        >
   580→          {expanded ? (
   581→            <ChevronDown className="size-3.5" />
   582→          ) : (
   583→            <ChevronRight className="size-3.5" />
   584→          )}
   585→        </button>
   586→
   587→        {isRenaming ? (
   588→          <Input
   589→            value={renameComparisonValue}
   590→            onChange={(e) => onRenameComparisonValueChange(e.target.value)}
   591→            onBlur={onCommitComparisonRename}
